# === kzwr_login_turnstile.py (Turnstile 自动登录, CloakBrowser) ===
# 注意: CloakBrowser 0.5.8 自带 Linux Chromium 146 有打包缺陷,
#       在飞牛(fnOS)等 Linux 服务器上不可用, 请改用 kzwr_login_camoufox.py
# cloakbrowser==0.5.8

# === kzwr_login_camoufox.py (Turnstile 自动登录, Camoufox) ===
# Camoufox 基于源码级反检测 Firefox, Linux 服务器上可用, 反检测更强。
# 安装后还需: python -m camoufox fetch   (下载浏览器二进制)
# 注意: 在飞牛上需用国内镜像安装, 见 README
camoufox[geoip]

# camoufox 0.5.5 依赖 playwright<1.61, 若与 cloakbrowser 同装会冲突。
# 推荐: 用 camoufox 方案时卸载 cloakbrowser 或仅装 camoufox。
# 登录脚本隐式调用 playwright.sync_api。
playwright==1.60.0
